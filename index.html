from ongo_memory import Memory

mem = Memory(user="f7a2")

# write
mem.episode(
  topic="calendar",
  text="call w/ Joonatan 4pm",
  vec=embed(...))

# read
hits = mem.recall(
  "what was I doing yesterday?",
  k=5)
